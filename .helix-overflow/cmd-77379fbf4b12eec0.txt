function run(command) {
  console.log("[helix] Running " + check + ": " + command);
  const r = spawnSync(command, { stdio: "inherit", shell: true });
  if (r.error) return 127;

}

function detectTestCommand(pkg) {
  if (pkg?.scripts?.test) return pkg.scripts.test;
  if (hasDep(pkg, "vitest")) return "vitest run";
  if (hasDep(pkg, "jest")) return "jest";
  if (hasDep(pkg, "mocha")) return "mocha";
  if (hasDep(pkg, "cypress")) return "cypress run";
  if (hasDep(pkg, "playwright") && exists("playwright.config.js", "playwright.config.ts")) return "playwright test";
  if (hasDep(pkg, "ava")) return "ava";
  if (hasDep(pkg, "@cucumber/cucumber")) return "cucumber-js";
  return null;
}

function detectLintCommand(pkg) {

function addScript(pkg, scriptName, command) {
  if (!pkg.scripts) pkg.scripts = {};
  if (pkg.scripts[scriptName]) return false;
  pkg.scripts[scriptName] = command;
  writeJson("package.json", pkg);
  console.log('[helix] Added package.json script "' + scriptName + '": "' + command + '"');
  return true;
}

    return run(command);

    if (scripts.typecheck || scripts["type-check"] || scripts.tsc) {
      return run(scripts.typecheck || scripts["type-check"] || scripts.tsc);

    if (hasWorkspaces) return runNpmScriptInWorkspaces("typecheck");
    const command = detectTypecheckCommand(pkg);
    if (!command) { console.log("[helix] No typecheck config or script found; skipping typecheck"); return 0; }
    if (autoFix && !hasWorkspaces) addScript(pkg, "typecheck", command);
    return run(command);

    if (scripts.build) return run(scripts.build);

    if (autoFix && !hasWorkspaces) addScript(pkg, "build", command);
    return run(command);

  console.log("[helix] Unknown check: " + check);

function ensurePytest() {
  const py = pythonCmd();
  const r = spawnSync(py + " -m pytest --version", { shell: true, stdio: "pipe" });
  if (r.status !== 0) spawnSync(py + " -m pip install pytest pytest-asyncio", { stdio: "inherit", shell: true });
  // ensure timeout plugin is available; ignore failures (e.g. no network)
  spawnSync(py + " -m pip install pytest-timeout", { stdio: "ignore", shell: true });
}

function pytestTimeoutFlag(py) {
  const r = spawnSync(py + " -m pytest --help", { shell: true, stdio: "pipe" });
  if (r.status === 0 && (r.stdout || "").toString().includes("--timeout")) return " --timeout=120";
  return "";
}

function startComposeIfNeeded() {
  const files = ["docker-compose.yml", "docker-compose.yaml", "compose.yml", "compose.yaml"].filter(f => exists(f));
  if (files.length === 0) return;
  const fileFlag = files.length > 1 ? " -f " + files.join(" -f ") : "";
  console.log("[helix] Found compose file(s): " + files.join(", "));
  const cmd = commandAvailable("docker-compose") ? "docker-compose" : (commandAvailable("docker") ? "docker compose" : null);
  if (!cmd) {
    console.log("[helix] Docker Compose not available; skipping service startup");
    return;
  }
  spawnSync(cmd + fileFlag + " up -d", { stdio: "inherit", shell: true });
  // Give services a moment to become reachable before tests connect.
  spawnSync("sleep 5", { shell: true, stdio: "ignore" });
}
